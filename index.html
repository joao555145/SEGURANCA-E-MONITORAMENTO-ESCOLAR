Bem-vindo ao sistema de monitoramento

Sensor PIR

O sensor PIR detecta movimentos no ambiente através da radiação infravermelha. Quando há movimento, o sistema ativa um LED e envia um alerta no monitor serial.

Detector de Fumaça

O detector de fumaça identifica gases e fumaça no ambiente utilizando o sensor MQ-2. Ao detectar risco, o sistema ativa um LED vermelho e um buzzer de alerta.

SENSOR DE MOVIMENTO

Esse é o nosso sensor de movimento e ele é relativamente fácil de montar e entender. Como o nome já diz, ele capta o movimento da pessoa ou objeto e o LED aciona na protoboard, e na ARDUINO IDE é mostrado no monitor serial "MOVIMENTO DETECTADO" ou "SEM MOVIMENTO". Bom, é mais fácil na prática! O código está aí, agora é só testar! 


#define SENSOR_PIR 2
#define LED 3
void setup() {
  pinMode(SENSOR_PIR, INPUT);
  pinMode(LED, OUTPUT);
  Serial.begin(9600);
}
void loop() {
  bool estado_sensor =
    digitalRead(SENSOR_PIR);
  if (estado_sensor == HIGH) {
    digitalWrite(LED, HIGH);
    Serial.println(
      "MOVIMENTO DETECTADO ✓"
    );
  } else {
    digitalWrite(LED, LOW);
    Serial.println(
      "SEM MOVIMENTO ☓"
    );
  }
  delay(200);
}

ultimo_estado = 0

while True:

    estado_sensor = sensor_pir.value()

    if estado_sensor == 1 and ultimo_estado == 0:

        print("🚨 MOVIMENTO DETECTADO ✓")
        print("Presença identificada!\n")

    elif estado_sensor == 0 and ultimo_estado == 1:

        print("✅ SEM MOVIMENTO ✕")
        print("Ambiente normal!\n")

    ultimo_estado = estado_sensor
    time.sleep(0.2)

DETECTOR DE FUMAÇA

Este é o nosso detector de fumaça. Em comparação com o detector de movimento, ele é um pouco mais complexo de montar, mas nada impossível! O dispositivo detecta a presença de gases e fumaça no ambiente e aciona um alarme sonoro e visual. Caso seja detectado gás ou fumaça na área, um LED vermelho será aceso indicando: "⚠️ GÁS DETECTADO!" Se o ambiente estiver limpo, um LED verde acenderá com a mensagem: "✅ Ambiente Seguro" Observação: A imagem foi retirada de um site oficial. 


#define MQ_analogico A0
#define MQ_digital 4
#define Buzzer 5
#define Led_Verde 6
#define Led_Vermelho 7

int valor_analogico;
int valor_digital;

// Limite do sensor

int limite_gas = 100;
void setup() {
  Serial.begin(9600);
  pinMode(MQ_analogico, INPUT);
  pinMode(MQ_digital, INPUT);
  pinMode(Buzzer, OUTPUT);
  pinMode(Led_Verde, OUTPUT);
  pinMode(Led_Vermelho, OUTPUT);
  digitalWrite(Led_Verde, HIGH);
  digitalWrite(Led_Vermelho, LOW);
  noTone(Buzzer);
}

void loop() {
  valor_analogico =
    analogRead(MQ_analogico);
  valor_digital =
    digitalRead(MQ_digital);

  // Mostra o nível no monitor serial

  Serial.print("Nivel de Gas/Fumaca: ");
  Serial.println(valor_analogico);
  if (valor_analogico > limite_gas) {
    digitalWrite(
      Led_Vermelho,
      HIGH
    );

    digitalWrite(
      Led_Verde,
      LOW
    );

    tone(Buzzer, 1000);
    Serial.print(
      "⚠️ AMBIENTE EM RISCO! | Nivel: "
    );
    Serial.println(valor_analogico);
  } else {
    digitalWrite(
      Led_Vermelho,
      LOW
    );
    digitalWrite(
      Led_Verde,
      HIGH
    );
    noTone(Buzzer);
    Serial.print(
      "✅ Ambiente Seguro | Nivel: "
    );
    Serial.println(valor_analogico);
  }
  delay(200);
}
